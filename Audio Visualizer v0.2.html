        function initThree() {

        // --- Shader Material ---

        const vertexShader = `

            uniform float uTime;

            uniform float uAudioLevel; // 0.0 to 1.0

            uniform float uComplexity; // 0.0 to 1.0 (roughness/jaggedness)

            

            varying vec2 vUv;

            varying vec3 vNormal;

            varying vec3 vPosition;

            varying float vDisplacement;


            // Simplex Noise function (simplified)

            vec3 mod289(vec3 x) { return x - floor(x * (1.0 / 289.0)) * 289.0; }

            vec4 mod289(vec4 x) { return x - floor(x * (1.0 / 289.0)) * 289.0; }

            vec4 permute(vec4 x) { return mod289(((x*34.0)+1.0)*x); }

            vec4 taylorInvSqrt(vec4 r) { return 1.79284291400159 - 0.85373472095314 * r; }

            float snoise(vec3 v) {

                const vec2  C = vec2(1.0/6.0, 1.0/3.0) ;

                const vec4  D = vec4(0.0, 0.5, 1.0, 2.0);

                vec3 i  = floor(v + dot(v, C.yyy) );

                vec3 x0 = v - i + dot(i, C.xxx) ;

                vec3 g = step(x0.yzx, x0.xyz);

                vec3 l = 1.0 - g;

                vec3 i1 = min( g.xyz, l.zxy );

                vec3 i2 = max( g.xyz, l.zxy );

                vec3 x1 = x0 - i1 + C.xxx;

                vec3 x2 = x0 - i2 + C.yyy;

                vec3 x3 = x0 - D.yyy;

                i = mod289(i);

                vec4 p = permute( permute( permute(

                            i.z + vec4(0.0, i1.z, i2.z, 1.0 ))

                        + i.y + vec4(0.0, i1.y, i2.y, 1.0 ))

                        + i.x + vec4(0.0, i1.x, i2.x, 1.0 ));

                float n_ = 0.142857142857;

                vec3  ns = n_ * D.wyz - D.xzx;

                vec4 j = p - 49.0 * floor(p * ns.z * ns.z);

                vec4 x_ = floor(j * ns.z);

                vec4 y_ = floor(j - 7.0 * x_ );

                vec4 x = x_ *ns.x + ns.yyyy;

                vec4 y = y_ *ns.x + ns.yyyy;

                vec4 h = 1.0 - abs(x) - abs(y);

                vec4 b0 = vec4( x.xy, y.xy );

                vec4 b1 = vec4( x.zw, y.zw );

                vec4 s0 = floor(b0)*2.0 + 1.0;

                vec4 s1 = floor(b1)*2.0 + 1.0;

                vec4 sh = -step(h, vec4(0.0));

                vec4 a0 = b0.xzyw + s0.xzyw*sh.xxyy ;

                vec4 a1 = b1.xzyw + s1.xzyw*sh.zzww ;

                vec3 p0 = vec3(a0.xy,h.x);

                vec3 p1 = vec3(a0.zw,h.y);

                vec3 p2 = vec3(a1.xy,h.z);

                vec3 p3 = vec3(a1.zw,h.w);

                vec4 norm = taylorInvSqrt(vec4(dot(p0,p0), dot(p1,p1), dot(p2, p2), dot(p3,p3)));

                p0 *= norm.x;

                p1 *= norm.y;

                p2 *= norm.z;

                p3 *= norm.w;

                vec4 m = max(0.6 - vec4(dot(x0,x0), dot(x1,x1), dot(x2,x2), dot(x3,x3)), 0.0);

                m = m * m;

                return 42.0 * dot( m*m, vec4( dot(p0,x0), dot(p1,x1), dot(p2,x2), dot(p3,x3) ) );

            }


            void main() {

                vUv = uv;

                vNormal = normal;

                

                // Base position

                vec3 pos = position;

                

                // Calculate noise based on position and time

                // Higher complexity = higher frequency noise, more jagged

                float noiseFreq = 1.0 + (uComplexity * 4.0); 

                float noiseAmp = uComplexity * 2.0; // How far vertices move

                

                float noiseVal = snoise(pos * noiseFreq + uTime * 0.5);

                

                // Displace along normal

                float displacement = noiseVal * noiseAmp;

                

                // Add audio reactivity to displacement (pulsing)

                displacement += uAudioLevel * 2.0; 


                vDisplacement = displacement;


                pos += normal * displacement;


                vPosition = pos;

                gl_Position = projectionMatrix * modelViewMatrix * vec4(pos, 1.0);

            }

        `;


        const fragmentShader = `

            uniform float uTime;

            uniform float uAudioLevel;

            uniform float uColorShift;

            

            varying vec2 vUv;

            varying vec3 vNormal;

            varying vec3 vPosition;

            varying float vDisplacement;


            void main() {

                // Base Color based on Audio Level and UVs

                vec3 colorA = vec3(0.1, 0.0, 0.2); // Deep Purple

                vec3 colorB = vec3(0.0, 0.8, 1.0); // Cyan

                vec3 colorC = vec3(1.0, 0.0, 0.5); // Hot Pink

                

                // Mix colors based on audio and position

                float mixVal = vDisplacement * 0.5 + uAudioLevel;

                

                vec3 finalColor = mix(colorA, colorB, mixVal);

                finalColor = mix(finalColor, colorC, uAudioLevel * abs(sin(uTime + vPosition.x)));

                

                // Add Color Shift from UI

                finalColor = fract(finalColor + uColorShift);


                // Fresnel effect (rim lighting) for 3D pop

                // Simple rim lighting based on normal z (approximation)

                float fresnel = pow(1.0 - abs(dot(vNormal, vec3(0,0,1))), 3.0);

                

                finalColor += vec3(fresnel * 0.5);


                gl_FragColor = vec4(finalColor, 1.0);

            }

        `;


        function createSpheres() {

            const geometry = new THREE.IcosahedronGeometry(1, 20); // High detail for displacement


            // Create 21 spheres in a line

            const totalWidth = (CONFIG.bandCount - 1) * CONFIG.spacing;

            const startX = -totalWidth / 2;


            for (let i = 0; i < CONFIG.bandCount; i++) {

                const material = new THREE.ShaderMaterial({

                    uniforms: {

                        uTime: uniforms.uTime,

                        uAudioLevel: { value: 0.0 },

                        uComplexity: { value: 0.0 },

                        uColorShift: uniforms.uColorShift

                    },

                    vertexShader: vertexShader,

                    fragmentShader: fragmentShader,

                    wireframe: false

                });


                const sphere = new THREE.Mesh(geometry, material);

                

                // Position

                sphere.position.x = startX + (i * CONFIG.spacing);

                sphere.position.y = 0;

                sphere.position.z = 0;


                // Store metadata

                sphere.userData = {

                    id: i,

                    baseScale: 1.0,

                    currentScale: 1.0,

                    targetScale: 1.0,

                    currentComplexity: 0,

                    targetComplexity: 0

                };


                scene.add(sphere);

                spheres.push(sphere);

            }

        }


        function createStars() {

            const geometry = new THREE.BufferGeometry();

            const vertices = [];

            for (let i = 0; i < 2000; i++) {

                vertices.push(

                    THREE.MathUtils.randFloatSpread(400),

                    THREE.MathUtils.randFloatSpread(400),

                    THREE.MathUtils.randFloatSpread(400)

                );

            }

            geometry.setAttribute('position', new THREE.Float32BufferAttribute(vertices, 3));

            const material = new THREE.PointsMaterial({ color: 0x888888, size: 0.7, transparent: true, opacity: 0.6 });

            const stars = new THREE.Points(geometry, material);

            scene.add(stars);

        }


        // --- Audio Logic ---


        async function startAudio() {

            try {

                state.audioContext = new (window.AudioContext || window.webkitAudioContext)();

                

                // Request Mic

                const stream = await navigator.mediaDevices.getUserMedia({ audio: true, video: false });

                

                state.microphone = state.audioContext.createMediaStreamSource(stream);

                state.analyser = state.audioContext.createAnalyser();

                

                state.analyser.fftSize = CONFIG.fftSize;

                state.analyser.smoothingTimeConstant = state.smoothing;

                

                state.microphone.connect(state.analyser);

                

                const bufferLength = state.analyser.frequencyBinCount;

                state.dataArray = new Uint8Array(bufferLength);


                // UI Updates

                state.isPlaying = true;

                startScreen.classList.remove('visible');

                startScreen.classList.add('hidden');

                controlsDiv.classList.remove('hidden');

                controlsDiv.classList.add('visible');

                statsPanel.classList.remove('hidden');


            } catch (err) {

                console.error(err);

                errorMsg.textContent = "Error: " + err.message + ". Please allow microphone access.";

                errorMsg.classList.remove('hidden');

            }

        }


        function stopAudio() {

            if (state.audioContext) {

                state.audioContext.close();

            }

            state.isPlaying = false;

            startScreen.classList.remove('hidden');

            startScreen.classList.add('visible');

            controlsDiv.classList.remove('visible');

            controlsDiv.classList.add('hidden');

            statsPanel.classList.add('hidden');

            

            // Reset spheres

            spheres.forEach(sphere => {

                sphere.userData.targetScale = 1;

                sphere.userData.targetComplexity = 0;

            });

        }

            return state.dataArray;

        }


        function mapFreqToBand(i) {

            // Logarithmic mapping of frequency bins to 21 bands

            const nyquist = state.audioContext.sampleRate / 2;

            const minLog = Math.log2(CONFIG.minFreq);

            const maxLog = Math.log2(CONFIG.maxFreq);

            const logRange = maxLog - minLog;

            

            const bandLog = minLog + (i / (CONFIG.bandCount - 1)) * logRange;

            const freq = Math.pow(2, bandLog);

            

            const bin = Math.floor((freq / nyquist) * (state.dataArray.length));

            return Math.min(bin, state.dataArray.length - 1);

        }


        function animate() {

            requestAnimationFrame(animate);


            state.time += 0.01;

            uniforms.uTime.value = state.time;


            if (state.isPlaying && state.dataArray) {

                const data = getFrequencyData();

                let totalVol = 0;

                let maxVol = 0;


                spheres.forEach((sphere, i) => {

                    const binIndex = mapFreqToBand(i);

                    let val = 0;

                    const range = 2; 

                    for(let k = -range; k <= range; k++) {

                        let idx = Math.max(0, Math.min(data.length-1, binIndex + k));

                        val += data[idx];

                    }

                    val = val / (range * 2 + 1);

                    

                    const normVal = val / 255;

                    totalVol += normVal;

                    if(normVal > maxVol) maxVol = normVal;


                    const targetScale = CONFIG.baseRadius + (normVal * 4 * state.sensitivity);

                    sphere.userData.targetScale = targetScale;

                    const targetComplexity = Math.min(1.0, normVal * 1.5); 

                    sphere.userData.targetComplexity = targetComplexity;


                    sphere.material.uniforms.uAudioLevel.value = normVal;

                    sphere.material.uniforms.uComplexity.value = sphere.userData.currentComplexity;

                    

                    sphere.scale.setScalar(

                        THREE.MathUtils.lerp(sphere.scale.x, sphere.userData.targetScale, 0.1)

                    );

                    

                    sphere.userData.currentComplexity = THREE.MathUtils.lerp(

                        sphere.userData.currentComplexity, 

                        sphere.userData.targetComplexity, 

                        0.05

                    );

                });


                const avgVol = totalVol / CONFIG.bandCount;

                document.getElementById('vol-counter').innerText = avgVol.toFixed(2);

                document.getElementById('comp-counter').innerText = maxVol.toFixed(2);

            } else {

                spheres.forEach((sphere, i) => {

                    const idleScale = CONFIG.baseRadius + Math.sin(state.time + i) * 0.2;

                    sphere.scale.setScalar(THREE.MathUtils.lerp(sphere.scale.x, idleScale, 0.05));

                    sphere.material.uniforms.uAudioLevel.value = THREE.MathUtils.lerp(sphere.material.uniforms.uAudioLevel.value, 0, 0.05);

                    sphere.material.uniforms.uComplexity.value = THREE.MathUtils.lerp(sphere.material.uniforms.uComplexity.value, 0, 0.05);

                });

            }


            // --- OPTIMIZED CAMERA MOTION ---

            // 3D Lissajous Figure-8 Pattern

            const t = state.time * CONFIG.cameraSpeed;

            const camRadius = 45; // Distance from center

            const camHeight = 35; // Maximum height (top/bottom)

            const camWidth = 20;  // Sway width (left/right)


            // Z (Depth): Moves Front (+) -> Back (-) -> Front (+)

            camera.position.z = Math.cos(t) * camRadius;


            // Y (Height): Moves Center -> Top (+) -> Center -> Bottom (-)

            // We use sin(t) here because when cos(t) is 1 (front), sin(t) is 0 (center).

            // When cos(t) is 0 (middle depth), sin(t) is 1 (top).

            camera.position.y = Math.sin(t) * camHeight;


            // X (Width): Adds the "Figure 8" twist.

            // By using sin(2*t), we oscillate left/right twice as fast, creating the crossover loop.

            camera.position.x = Math.sin(t * 2) * camWidth;


            // Ensure camera always looks at the center of the spheres

            camera.lookAt(0, 0, 0);


            renderer.render(scene, camera);

        }


        function onWindowResize() {

            camera.aspect = window.innerWidth / window.innerHeight;

            camera.updateProjectionMatrix();

            renderer.setSize(window.innerWidth, window.innerHeight);

        }
